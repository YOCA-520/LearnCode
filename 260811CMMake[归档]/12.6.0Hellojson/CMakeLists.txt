# ============================================================
# CMakeLists.txt —— 本项目的 CMake 构建脚本
# CMake 根据这个文件决定：用什么编译器、编哪些源码、链接哪些库、
# 最终生成什么可执行文件。整个构建流程都由它"指挥"。
# 文件里的注释以 # 开头，会被 CMake 忽略，只给人看。
# ============================================================

# ------------------------------------------------------------
# 1. 最低 CMake 版本要求
#    声明本脚本至少需要 CMake 3.10 才能正确解析。
#    如果本机 CMake 版本低于 3.10，配置时会直接报错，
#    防止用太老的特性不兼容。这是每个 CMakeLists.txt 的第一行。
# ------------------------------------------------------------
cmake_minimum_required(VERSION 3.25)

# ------------------------------------------------------------
# 2. 工程声明
#    project() 会做两件事：
#    ① 给整个工程起名字（HelloJson）；
#    ② 自动生成几个变量，其中最常用的是 ${PROJECT_NAME}，
#       值就是工程名 "HelloJson"。
#    后续代码里用 ${PROJECT_NAME} 而不是写死 "HelloJson"，
#    这样以后改工程名只改这一处即可，避免遗漏。
# ------------------------------------------------------------
project(HelloJson)

# ------------------------------------------------------------
# 3. 指定 C++ 标准
#    下面三个 set() 配合使用，固定使用 C++20：
# ------------------------------------------------------------
# 编译时加 -std=c++20，告诉编译器用 C++20 的语法特性
set(CMAKE_CXX_STANDARD 20)
# REQUIRED = ON 表示"必须"用 C++20，
# 如果编译器不支持 C++20，配置阶段就直接报错，而不是悄悄降级
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 关闭编译器扩展（如 GNU 的私有扩展），
# 保证代码只用标准 C++20，换编译器也能编译
set(CMAKE_CXX_EXTENSIONS OFF)

# ------------------------------------------------------------
# 4. 设置可执行文件生成位置
#    CMAKE_RUNTIME_OUTPUT_DIRECTORY 决定"运行时文件"（可执行文件）
#    输出到哪里。${CMAKE_BINARY_DIR} 是构建目录（就是 build/），
#    拼上 /bin 表示生成到 build/bin/ 目录下，方便统一管理。
#    重要：必须写在 add_executable() 之前才生效，
#    因为输出目录在创建目标那一刻就定好了。
# ------------------------------------------------------------
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# ------------------------------------------------------------
# 5. 查找第三方库：nlohmann_json
#    find_package() 会在系统/vcpkg 等地方寻找已经安装好的库，
#    找到后把它的头文件路径、库路径都配置好，
#    并提供一个"库目标" nlohmann_json::nlohmann_json 供链接使用。
#    REQUIRED 表示"找不到就报错"，避免程序在缺依赖时悄悄编译失败。
#    本例中 nlohmann_json 是通过 vcpkg 安装的（3.12.0），
#    配置时由 vcpkg 的 toolchain 文件自动定位。
# ------------------------------------------------------------
find_package(nlohmann_json REQUIRED)

# ------------------------------------------------------------
# 6. 生成可执行文件
#    add_executable(目标名 源文件...)
#    告诉 CMake：把 12.6.0JSONbasicExample示例.cpp 编译链接成
#    名为 ${PROJECT_NAME}（即 HelloJson）的可执行文件。
#    多个源文件用空格分隔继续往后写即可。
# ------------------------------------------------------------
add_executable(${PROJECT_NAME} 12.6.0JSONbasicExample示例.cpp)



# ------------------------------------------------------------
# 7. 链接库
#    target_link_libraries(目标名 作用域 库...)
#    把上一步的 nlohmann_json::nlohmann_json 链接到 HelloJson 上，
#    这样源码里 #include <nlohmann/json.hpp> 才能找到头文件。
#    PRIVATE 表示：这个依赖只在 HelloJson 内部使用，
#    不会被传递给其他依赖 HelloJson 的工程（封装性好）。
# ------------------------------------------------------------
target_link_libraries(${PROJECT_NAME} PRIVATE nlohmann_json::nlohmann_json)
